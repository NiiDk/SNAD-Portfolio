# Multi-Tenant Offline Sync Resolver with Conflict Management
class POSSyncResolver:
    @transaction.atomic
    def process_offline_batch(self, tenant_id, device_id, transactions_payload):
        tenant = Tenant.objects.select_for_update().get(id=tenant_id)
        synced_records = []
        for tx in transactions_payload:
            # Deterministic idempotency key verification
            if TransactionLedger.objects.filter(idempotency_key=tx['idempotency_key']).exists():
                continue
            record = TransactionLedger.objects.create(
                tenant=tenant, device_id=device_id,
                amount=tx['amount'], payment_channel=tx['channel'],
                timestamp=tx['client_timestamp'], status='SETTLED'
            )
            # Trigger background inventory reconciliation via Celery
            reconcile_inventory_batch.delay(tenant_id, tx['items'])
            synced_records.append(record.id)
        return {"status": "SUCCESS", "synced_count": len(synced_records)}

// Real-time POS Cart State with Optimistic UI & Webhook Telemetry
export function POSCheckoutEngine({ storeId }: { storeId: string }) {
  const { cart, total, processPayment } = usePOSStore();
  const [status, setStatus] = useState<'idle' | 'processing' | 'printed'>('idle');

  const handleCheckout = async (paymentType: 'CASH' | 'MOMO') => {
    setStatus('processing');
    const res = await processPayment({ storeId, items: cart, total, paymentType });
    if (res.ok) {
      await ThermalPrinterBridge.printReceipt(res.receiptPayload);
      setStatus('printed');
    }
  };

  return <POSCartInterface total={total} onCheckout={handleCheckout} status={status} />;
}

// Direct Bluetooth ESC/POS Thermal Printer Protocol Implementation
class ThermalPrinterService {
  final BluetoothPrinterManager _manager = BluetoothPrinterManager();

  Future<bool> printCustomerReceipt(ReceiptData receipt) async {
    final List<int> bytes = [];
    final generator = Generator(PaperSize.mm58, await CapabilityProfile.load());
    
    bytes.addAll(generator.text(receipt.storeName, styles: PosStyles(align: PosAlign.center, bold: true)));
    bytes.addAll(generator.hr());
    for (var item in receipt.items) {
      bytes.addAll(generator.row([
        PosColumn(text: item.name, width: 8),
        PosColumn(text: "GH₵${item.price}", width: 4, styles: PosStyles(align: PosAlign.right)),
      ]));
    }
    bytes.addAll(generator.cut());
    return await _manager.writeBytes(bytes);
  }
}
